
@startuml
skinparam classAttributeIconSize 0

class User {
  - id : int
  - name : string
  - email : string
  - phone_number : string
  - password : string
  - role : string <<admin, customer>>
}

class Coach {
  - id : int
  - name : string
  - specialty : string
  - years_experience : int
}

class Room {
  - id : int
  - name : string
  - type : string
  - capacity : int
  - price_per_night : float
  - total_stock : int
  - is_active : boolean
  - image_path : string
}

class Package {
  - id : int
  - name : string
  - description : text
  - duration_days : int
  - base_price : float
  - image_path : string
}

class Activity {
  - id : int
  - name : string
  - duration_minutes : int
}

class PackageActivity <<Pivot>> {
  - included_sessions : int
}

class Event {
  - id : int
  - title : string
  - description : text
  - event_date : datetime
  - max_participants : int
  - price : float
  - image_path : string
}

class Booking {
  - id : int
  - start_date : date
  - end_date : date
  - total_price : float
  - payment_status : string <<pending, paid, failed>>
  - status : string <<pending, confirmed, finished, cancelled>>
}

class BookingItem <<Pivot>> {
  - quantity : int
}

class Review {
  - id : int
  - rating : int
  - comment : text
}

' Core Booking Relationships
User "1" -- "0..*" Booking : makes >
Booking "1" -- "1" Room : reserves >
Booking "1" -- "1" Package : requires >

' Review Logic (The Airbnb Method)
User "1" -- "0..*" Review : writes >
Booking "1" -- "0..1" Review : triggers >
Package "1" -- "0..*" Review : displays >

' Booking Extras (Events)
Booking "1" -- "0..*" BookingItem : contains >
BookingItem "0..*" -- "1" Event : references >

' Package Inclusions 
Package "1" -- "0..*" PackageActivity : includes >
PackageActivity "0..*" -- "1" Activity : references >

' The Coach Assignment
Coach "1" -- "0..*" Activity : leads >

@enduml
